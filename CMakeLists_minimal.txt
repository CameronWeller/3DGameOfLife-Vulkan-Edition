# Minimal CMakeLists.txt for 3D Game of Life - Vulkan Edition
cmake_minimum_required(VERSION 3.20)

project(3DGameOfLife-Vulkan-Edition-Minimal
    VERSION 1.0.0
    DESCRIPTION "Minimal 3D Conway's Game of Life with Vulkan"
    LANGUAGES CXX
)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# vcpkg integration
if(DEFINED ENV{VCPKG_ROOT} AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
endif()

# Find packages using vcpkg
find_package(Vulkan REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)

# Create minimal executable
# Note: This is a standalone minimal build that compiles sources directly
# (not using the unified library to avoid dependencies)
add_executable(minimal_vulkan_app
    src/main_minimal.cpp
    src/VulkanContext.cpp
    src/DeviceManager.cpp
    src/WindowManager.cpp
    src/Logger.cpp
    src/Camera.cpp
    src/engine/vulkan/resources/ShaderManager.cpp
    src/SimplePatternLoader.cpp
)

# Link libraries
target_link_libraries(minimal_vulkan_app PRIVATE
    Vulkan::Vulkan
    glfw
    glm::glm
)

# Include directories
target_include_directories(minimal_vulkan_app PRIVATE
    include
    include/engine
    include/engine/vulkan
    include/engine/vulkan/resources
    ${Vulkan_INCLUDE_DIRS}
)

# Compiler settings
if(MSVC)
    target_compile_options(minimal_vulkan_app PRIVATE /W4)
else()
    target_compile_options(minimal_vulkan_app PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Add definitions
target_compile_definitions(minimal_vulkan_app PRIVATE
    $<$<CONFIG:Debug>:DEBUG>
    $<$<CONFIG:Release>:NDEBUG>
)

# Shader compilation
find_program(GLSLC_EXECUTABLE glslc PATHS ${VULKAN_SDK}/bin ENV PATH)
if(GLSLC_EXECUTABLE)
    set(SHADER_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
    set(SHADER_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/shaders)
    file(MAKE_DIRECTORY ${SHADER_BINARY_DIR})
    
    # Define shader output files
    set(COMPUTE_SHADER_OUTPUT ${SHADER_BINARY_DIR}/game_of_life_3d.comp.spv)
    set(VERTEX_SHADER_OUTPUT ${SHADER_BINARY_DIR}/minimal.vert.spv)
    set(FRAGMENT_SHADER_OUTPUT ${SHADER_BINARY_DIR}/minimal.frag.spv)
    
    # Compile compute shader
    add_custom_command(
        OUTPUT ${COMPUTE_SHADER_OUTPUT}
        COMMAND ${GLSLC_EXECUTABLE} -o ${COMPUTE_SHADER_OUTPUT} ${SHADER_SOURCE_DIR}/game_of_life_3d.comp
        DEPENDS ${SHADER_SOURCE_DIR}/game_of_life_3d.comp
        COMMENT "Compiling compute shader: game_of_life_3d.comp"
    )
    
    # Compile vertex/fragment shaders
    add_custom_command(
        OUTPUT ${VERTEX_SHADER_OUTPUT} ${FRAGMENT_SHADER_OUTPUT}
        COMMAND ${GLSLC_EXECUTABLE} -o ${VERTEX_SHADER_OUTPUT} ${SHADER_SOURCE_DIR}/minimal.vert
        COMMAND ${GLSLC_EXECUTABLE} -o ${FRAGMENT_SHADER_OUTPUT} ${SHADER_SOURCE_DIR}/minimal.frag
        DEPENDS ${SHADER_SOURCE_DIR}/minimal.vert ${SHADER_SOURCE_DIR}/minimal.frag
        COMMENT "Compiling graphics shaders"
    )
    
    # Create custom target that depends on all shader outputs
    # This ensures shaders are compiled before the executable is built
    add_custom_target(compile_shaders_minimal
        DEPENDS 
            ${COMPUTE_SHADER_OUTPUT}
            ${VERTEX_SHADER_OUTPUT}
            ${FRAGMENT_SHADER_OUTPUT}
        COMMENT "Compiling all shaders for minimal build"
    )
    
    # Make executable depend on shader compilation target
    add_dependencies(minimal_vulkan_app compile_shaders_minimal)
    
    # Copy shaders to output directory after build
    add_custom_command(TARGET minimal_vulkan_app POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:minimal_vulkan_app>/shaders
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${COMPUTE_SHADER_OUTPUT} $<TARGET_FILE_DIR:minimal_vulkan_app>/shaders/
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${VERTEX_SHADER_OUTPUT} $<TARGET_FILE_DIR:minimal_vulkan_app>/shaders/
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FRAGMENT_SHADER_OUTPUT} $<TARGET_FILE_DIR:minimal_vulkan_app>/shaders/
        COMMENT "Copying shaders to output directory"
    )
else()
    message(WARNING "glslc not found. Shaders must be compiled manually.")
endif()

message(STATUS "Minimal Vulkan App configured successfully")
