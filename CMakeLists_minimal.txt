# Minimal CMakeLists.txt for 3D Game of Life - Vulkan Edition
cmake_minimum_required(VERSION 3.20)

project(3DGameOfLife-Vulkan-Edition-Minimal
    VERSION 1.0.0
    DESCRIPTION "Minimal 3D Conway's Game of Life with Vulkan"
    LANGUAGES CXX
)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# vcpkg integration
if(DEFINED ENV{VCPKG_ROOT} AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake" CACHE STRING "")
endif()

# Find packages using vcpkg
find_package(Vulkan REQUIRED)
find_package(glfw3 CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)

# Create minimal executable
# Note: This is a standalone minimal build that compiles sources directly
# (not using the unified library to avoid dependencies)
add_executable(minimal_vulkan_app
    src/main_minimal.cpp
    src/VulkanContext.cpp
    src/DeviceManager.cpp
    src/WindowManager.cpp
    src/Logger.cpp
    src/engine/vulkan/resources/ShaderManager.cpp
)

# Link libraries
target_link_libraries(minimal_vulkan_app PRIVATE
    Vulkan::Vulkan
    glfw
    glm::glm
)

# Include directories
target_include_directories(minimal_vulkan_app PRIVATE
    include
    include/engine
    include/engine/vulkan
    include/engine/vulkan/resources
    ${Vulkan_INCLUDE_DIRS}
)

# Compiler settings
if(MSVC)
    target_compile_options(minimal_vulkan_app PRIVATE /W4)
else()
    target_compile_options(minimal_vulkan_app PRIVATE -Wall -Wextra -Wpedantic)
endif()

# Add definitions (Vulkan uses a 0..1 depth range)
target_compile_definitions(minimal_vulkan_app PRIVATE
    GLM_FORCE_DEPTH_ZERO_TO_ONE
    $<$<CONFIG:Debug>:DEBUG>
    $<$<CONFIG:Release>:NDEBUG>
)

# Shader compilation
find_program(GLSLC_EXECUTABLE glslc PATHS ${VULKAN_SDK}/bin ENV PATH)
if(GLSLC_EXECUTABLE)
    set(SHADER_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/shaders)
    set(SHADER_BINARY_DIR ${CMAKE_CURRENT_BINARY_DIR}/shaders)
    file(MAKE_DIRECTORY ${SHADER_BINARY_DIR})
    
    # The minimal app (src/main_minimal.cpp) loads these from shaders/ next to the executable.
    set(MINIMAL_SPV)
    foreach(shader life3d_chunks.comp life3d_world.vert life3d_world.frag life3d_screen.vert life3d_screen.frag)
        set(spv ${SHADER_BINARY_DIR}/${shader}.spv)
        add_custom_command(
            OUTPUT ${spv}
            COMMAND ${GLSLC_EXECUTABLE} -o ${spv} ${SHADER_SOURCE_DIR}/${shader}
            DEPENDS ${SHADER_SOURCE_DIR}/${shader} ${SHADER_SOURCE_DIR}/life3d_frame.glsl
            COMMENT "Compiling shader: ${shader}"
        )
        list(APPEND MINIMAL_SPV ${spv})
    endforeach()

    add_custom_target(compile_shaders_minimal DEPENDS ${MINIMAL_SPV})
    add_dependencies(minimal_vulkan_app compile_shaders_minimal)

    add_custom_command(TARGET minimal_vulkan_app POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:minimal_vulkan_app>/shaders
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${MINIMAL_SPV} $<TARGET_FILE_DIR:minimal_vulkan_app>/shaders/
        COMMENT "Copying shaders to output directory"
    )
else()
    message(WARNING "glslc not found. Shaders must be compiled manually.")
endif()

message(STATUS "Minimal Vulkan App configured successfully")
