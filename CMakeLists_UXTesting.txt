cmake_minimum_required(VERSION 3.16)
project(UXTestingFramework)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Set output directories
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Find required packages
find_package(Threads REQUIRED)

# Include directories
include_directories(include)

# Source files for the framework
set(FRAMEWORK_SOURCES
    src/UXTestingFramework.cpp
    src/WindowsUXTesting.cpp
)

# Create framework library
add_library(UXTestingFramework STATIC ${FRAMEWORK_SOURCES})

# Link libraries for framework
target_link_libraries(UXTestingFramework
    Threads::Threads
)

# Windows-specific libraries
if(WIN32)
    target_link_libraries(UXTestingFramework
        user32
        gdi32
        kernel32
    )
endif()

# Test launcher executable
add_executable(UXTestLauncher src/UXTestLauncher.cpp)
target_link_libraries(UXTestLauncher UXTestingFramework)

# Minecraft test executable (placeholder)
add_executable(MinecraftUXTester src/MinecraftUXTester.cpp)
target_link_libraries(MinecraftUXTester UXTestingFramework)

# Installation
install(TARGETS UXTestingFramework UXTestLauncher MinecraftUXTester
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(DIRECTORY include/ DESTINATION include)

# Testing
enable_testing()
add_test(NAME BasicFrameworkTest COMMAND UXTestLauncher)

# Documentation
find_package(Doxygen)
if(DOXYGEN_FOUND)
    set(DOXYGEN_IN ${CMAKE_CURRENT_SOURCE_DIR}/docs/Doxyfile.in)
    set(DOXYGEN_OUT ${CMAKE_CURRENT_BINARY_DIR}/Doxyfile)
    
    configure_file(${DOXYGEN_IN} ${DOXYGEN_OUT} @ONLY)
    
    add_custom_target(docs ALL
        COMMAND ${DOXYGEN_EXECUTABLE} ${DOXYGEN_OUT}
        WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
        COMMENT "Generating API documentation with Doxygen"
        VERBATIM
    )
endif()

# Package configuration
include(CMakePackageConfigHelpers)
write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/UXTestingFrameworkConfigVersion.cmake"
    VERSION 1.0.0
    COMPATIBILITY AnyNewerVersion
)

configure_package_config_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/UXTestingFrameworkConfig.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/UXTestingFrameworkConfig.cmake"
    INSTALL_DESTINATION lib/cmake/UXTestingFramework
)

install(FILES
    "${CMAKE_CURRENT_BINARY_DIR}/UXTestingFrameworkConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/UXTestingFrameworkConfigVersion.cmake"
    DESTINATION lib/cmake/UXTestingFramework
)
